import * as THREE from 'three';


const scene = new THREE.Scene();

scene.background = new THREE.Color(0x87ceeb);

scene.fog = new THREE.Fog(0x87ceeb, 100, 260);


const camera = new THREE.PerspectiveCamera(

65, innerWidth/innerHeight, 0.1, 500

);

const renderer = new THREE.WebGLRenderer({

antialias:false, powerPreference:'low-power'

});

renderer.setPixelRatio(Math.min(devicePixelRatio,1.5));

renderer.setSize(innerWidth,innerHeight);

renderer.domElement.style.position='fixed';

renderer.domElement.style.inset='0';

document.body.prepend(renderer.domElement);


scene.add(new THREE.HemisphereLight(0xffffff,0x536c43,2));

const sun = new THREE.DirectionalLight(0xffffff,2);

sun.position.set(30,60,20);

scene.add(sun);


// 地面

const ground = new THREE.Mesh(

new THREE.PlaneGeometry(500,500),

new THREE.MeshStandardMaterial({color:0x43834c})

);

ground.rotation.x=-Math.PI/2;

ground.position.y=-0.15;

scene.add(ground);


// 楕円コース

const road = new THREE.Mesh(

new THREE.TorusGeometry(30,5,8,120),

new THREE.MeshStandardMaterial({color:0x424953,roughness:1})

);

road.rotation.x=Math.PI/2;

road.scale.set(1.5,1,1);

road.position.y=0.05;

scene.add(road);


// コースの内側の芝生

const inner = new THREE.Mesh(

new THREE.CircleGeometry(25,80),

new THREE.MeshStandardMaterial({color:0x4d9950})

);

inner.rotation.x=-Math.PI/2;

inner.scale.set(1.5,1,1);

inner.position.y=-0.08;

scene.add(inner);


// コースの外側に木を配置

for(let i=0;i<60;i++){

const a=i*Math.PI*2/60;

const x=64*Math.cos(a),z=45*Math.sin(a);

const trunk=new THREE.Mesh(

new THREE.CylinderGeometry(.35,.5,3,6),

new THREE.MeshStandardMaterial({color:0x77502d})

);

trunk.position.set(x,1.5,z);

scene.add(trunk);

const leaves=new THREE.Mesh(

new THREE.ConeGeometry(2.2,5,7),

new THREE.MeshStandardMaterial({color:0x176b3a})

);

leaves.position.set(x,5,z);

scene.add(leaves);

}


// カートを生成

function makeKart(color){

const g=new THREE.Group();

const mat=new THREE.MeshStandardMaterial({color,roughness:.55});

const body=new THREE.Mesh(new THREE.BoxGeometry(2,0.7,3),mat);

body.position.y=.8;

g.add(body);


const seat=new THREE.Mesh(

new THREE.BoxGeometry(1.1,.65,1),

new THREE.MeshStandardMaterial({color:0x222631})

);

seat.position.set(0,1.35,.25);

g.add(seat);


const head=new THREE.Mesh(

new THREE.SphereGeometry(.42,12,10),

new THREE.MeshStandardMaterial({color:0xffd5ad})

);

head.position.set(0,1.9,.2);

g.add(head);


for(const x of [-1,1]){

for(const z of [-.95,.95]){

const wheel=new THREE.Mesh(

new THREE.CylinderGeometry(.42,.42,.28,12),

new THREE.MeshStandardMaterial({color:0x17191d})

);

wheel.rotation.z=Math.PI/2;

wheel.position.set(x,.48,z);

g.add(wheel);

}

}

scene.add(g);

return g;

}


const me=makeKart(0x16c985);

const rival=makeKart(0xf0a03b);

rival.visible=false;


let angle=0, speed=0, steer=0, drift=0;

let boost=0, lap=1, lastAngle=0, totalAngle=0;

let peer=null, conn=null, online=false, started=false;

let remote={angle:Math.PI,speed:0,boost:0};

let sendTime=0;

const keys={left:false,right:false,gas:false,brake:false,drift:false};


function setStatus(s){

document.getElementById('status').textContent=s;

}

function bindHold(id,key){

const el=document.getElementById(id);

const down=e=>{e.preventDefault();keys[key]=true};

const up=e=>{e.preventDefault();keys[key]=false};

el.addEventListener('pointerdown',down);

el.addEventListener('pointerup',up);

el.addEventListener('pointercancel',up);

el.addEventListener('pointerleave',up);

}

bindHold('left','left');

bindHold('right','right');

bindHold('gas','gas');

if(e.key==='ArrowRight'||e.key==='d')keys.right=true;

if(e.key==='ArrowDown'||e.key==='s')keys.brake=true;

if(e.key==='Shift')keys.drift=true;

},{passive:false});

if(e.key==='Shift')keys.drift=false;

});

// 部屋を作る／相手と接続

function setupConnection(c){

conn=c;

c.on('open',()=>{

online=true;

setStatus('オンライン接続済み');

document.getElementById('start').style.display='block';

});

c.on('data',data=>{

if(data&&data.type==='state'){

remote=data.state;

rival.visible=true;

}

});

c.on('close',()=>{

online=false;

setStatus('相手との接続が切れました');

});

c.on('error',()=>setStatus('通信エラー'));

}

document.getElementById('solo').onclick=()=>{

document.getElementById('lobby').style.display='none';

setStatus('ソロレース');

};

document.getElementById('host').onclick=()=>{

if(!window.Peer){setStatus('通信ライブラリを読み込めません');return}

peer=new Peer();

setStatus('部屋を準備中...');

peer.on('open',id=>{

document.getElementById('myroom').textContent='部屋ID: '+id;

document.getElementById('room').value=id;

setStatus('相手の参加を待っています');

});

peer.on('connection',setupConnection);

peer.on('error',e=>setStatus('通信エラー: '+e.type));

};

document.getElementById('join').onclick=()=>{

const id=document.getElementById('room').value.trim();

if(!id)return setStatus('部屋IDを入力してください');

peer=new Peer();

setStatus('部屋に接続中...');

peer.on('open',()=>setupConnection(peer.connect(id)));

peer.on('error',e=>setStatus('接続できません: '+e.type));

};

document.getElementById('start').onclick=()=>{

if(!online)return setStatus('相手との接続を待っています');

started=true;

document.getElementById('lobby').style.display='none';

};


// カートの位置を楕円コースに沿って更新

function placeKart(k,a){

k.position.set(45*Math.cos(a),0,30*Math.sin(a));

const dx=-45*Math.sin(a);

const dz=30*Math.cos(a);

k.rotation.y=Math.atan2(dx,dz);

}

function animate(t){

requestAnimationFrame(animate);

const dt=Math.min(.04, (t-(animate.prev||t))/1000);

animate.prev=t;


if(started){

if(keys.gas)speed=Math.min(42,speed+24*dt);

else speed=Math.max(12,speed-5*dt);

if(keys.brake)speed=Math.max(0,speed-35*dt);


steer=(keys.right?1:0)-(keys.left?1:0);

angle+=steer*speed/35*dt*(keys.drift?1.55:1);

angle+=speed/35*dt;

if(boost>0){speed=Math.min(55,speed+25*dt);boost-=dt}


totalAngle+=speed/35*dt;

lap=1+Math.floor(totalAngle/(Math.PI*2));

placeKart(me,angle);

me.rotation.y+=steer*.15;


document.getElementById('speed').textContent=

Math.round(speed*4)+' km/h';

document.getElementById('lap').textContent='LAP '+lap;


if(online&&conn&&conn.open&&t-sendTime>60){

conn.send({type:'state',state:{angle,speed,boost}});

sendTime=t;

}

}


if(rival.visible){

placeKart(rival,remote.angle);

rival.rotation.y+=.12;

}


// カメラを自分のカートの後ろへ

const forward=new THREE.Vector3(

Math.sin(me.rotation.y),0,Math.cos(me.rotation.y)

);

const target=me.position.clone().add(new THREE.Vector3(0,2,0));

const behind=me.position.clone()

.addScaledVector(forward,-13)

.add(new THREE.Vector3(0,8,0));

camera.position.lerp(behind,.09);

camera.lookAt(target);


renderer.render(scene,camera);

}

requestAnimationFrame(animate);


addEventListener('resize',()=>{

camera.aspect=innerWidth/innerHeight;

camera.updateProjectionMatrix();

renderer.setSize(innerWidth,innerHeight);

});
